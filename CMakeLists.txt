cmake_minimum_required(VERSION 3.16)
project(heavyverb VERSION 0.1.0 LANGUAGES C CXX)
add_library(heavyverb INTERFACE)
target_include_directories(heavyverb INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(heavyverb INTERFACE cxx_std_17)
option(HEAVYVERB_PORTABLE_FFT "Use the dependency-free custom FFT with the shared partitioning policy" OFF)
if(HEAVYVERB_PORTABLE_FFT)
  target_compile_definitions(heavyverb INTERFACE HEAVYVERB_PORTABLE_FFT)
else()
  add_library(heavyverb_pffft STATIC third_party/pffft/pffft.c third_party/pffft/pffft_common.c)
  set_target_properties(heavyverb_pffft PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE ON)
  target_include_directories(heavyverb_pffft PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/pffft)
  target_compile_definitions(heavyverb_pffft PUBLIC PFFFT_STATIC_DEFINE)
  target_link_libraries(heavyverb INTERFACE heavyverb_pffft)
endif()
option(HEAVYVERB_UNIFORM_PARTITIONS "Use uniform partitions for comparison" OFF)
if(HEAVYVERB_UNIFORM_PARTITIONS)
  target_compile_definitions(heavyverb INTERFACE HEAVYVERB_UNIFORM_PARTITIONS)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(heavyverb_test tests/core.cpp)
  target_link_libraries(heavyverb_test PRIVATE heavyverb)
  add_test(NAME convolution COMMAND heavyverb_test)
  if(HEAVYVERB_PORTABLE_FFT)
    add_executable(heavyverb_fft_test tests/fft.cpp)
    target_link_libraries(heavyverb_fft_test PRIVATE heavyverb)
    add_test(NAME fft COMMAND heavyverb_fft_test)
    add_executable(heavyverb_fft_test_scalar tests/fft.cpp)
    target_compile_definitions(heavyverb_fft_test_scalar PRIVATE HEAVYVERB_FFT_SCALAR)
    target_link_libraries(heavyverb_fft_test_scalar PRIVATE heavyverb)
    add_test(NAME fft_scalar COMMAND heavyverb_fft_test_scalar)
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(heavyverb_realtime tests/realtime.cpp)
    target_link_libraries(heavyverb_realtime PRIVATE heavyverb)
    target_link_options(heavyverb_realtime PRIVATE "LINKER:--wrap=malloc,--wrap=calloc,--wrap=realloc")
    add_test(NAME realtime COMMAND heavyverb_realtime)
  endif()
endif()
add_executable(heavyverb_benchmark tests/benchmark.cpp)
target_link_libraries(heavyverb_benchmark PRIVATE heavyverb)
set(PD_INCLUDE_DIR "" CACHE PATH "Directory containing m_pd.h")
if(PD_INCLUDE_DIR)
  add_library(heavyverb_pd MODULE pd/heavyverb_tilde.cpp)
  target_include_directories(heavyverb_pd PRIVATE ${PD_INCLUDE_DIR})
  target_link_libraries(heavyverb_pd PRIVATE heavyverb)
  set_target_properties(heavyverb_pd PROPERTIES PREFIX "" OUTPUT_NAME "heavyverb~")
  if(APPLE)
    set_target_properties(heavyverb_pd PROPERTIES SUFFIX ".pd_darwin" LINK_FLAGS "-undefined dynamic_lookup")
  elseif(UNIX)
    set_target_properties(heavyverb_pd PROPERTIES SUFFIX ".pd_linux")
  else()
    message(FATAL_ERROR "Pd external packaging is currently supported on Linux/macOS only")
  endif()
endif()
