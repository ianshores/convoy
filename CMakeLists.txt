cmake_minimum_required(VERSION 3.16)
project(convoy VERSION 0.2.0 LANGUAGES CXX)
add_library(convoy INTERFACE)
add_library(convoy::convoy ALIAS convoy)
target_include_directories(convoy INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/vendor/dsp/include>
  $<INSTALL_INTERFACE:include>)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/dsp/include/dsp/fft.hpp")
  message(FATAL_ERROR "DSP dependency missing: run git submodule update --init --recursive")
endif()
target_compile_features(convoy INTERFACE cxx_std_17)
option(CONVOY_DISABLE_AVX "Disable runtime AVX acceleration in the custom backend" OFF)
if(CONVOY_DISABLE_AVX)
  target_compile_definitions(convoy INTERFACE CONVOY_DISABLE_AVX)
endif()
option(CONVOY_UNIFORM_PARTITIONS "Use uniform partitions for comparison" OFF)
if(CONVOY_UNIFORM_PARTITIONS)
  target_compile_definitions(convoy INTERFACE CONVOY_UNIFORM_PARTITIONS)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(convoy_cutoff tests/cutoff.cpp)
  target_link_libraries(convoy_cutoff PRIVATE convoy)
  add_test(NAME cutoff COMMAND convoy_cutoff)
  add_executable(convoy_test tests/core.cpp)
  target_link_libraries(convoy_test PRIVATE convoy)
  add_test(NAME convolution COMMAND convoy_test)
  # All builds use the custom FFT.
  add_executable(convoy_fft_test tests/fft.cpp)
  target_link_libraries(convoy_fft_test PRIVATE convoy)
  add_test(NAME fft COMMAND convoy_fft_test)
  add_executable(convoy_fft_test_scalar tests/fft.cpp)
  target_compile_definitions(convoy_fft_test_scalar PRIVATE CONVOY_FFT_SCALAR)
  target_link_libraries(convoy_fft_test_scalar PRIVATE convoy)
  add_test(NAME fft_scalar COMMAND convoy_fft_test_scalar)
  foreach(mode half packed)
    add_executable(convoy_fft_${mode} tests/fft.cpp)
    target_link_libraries(convoy_fft_${mode} PRIVATE convoy)
    if(mode STREQUAL "half")
      target_compile_definitions(convoy_fft_${mode} PRIVATE CONVOY_TEST_HALF_FORWARD)
    else()
      target_compile_definitions(convoy_fft_${mode} PRIVATE CONVOY_TEST_PACKED CONVOY_TEST_PACKED8)
    endif()
    add_test(NAME fft_${mode} COMMAND convoy_fft_${mode})
  endforeach()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(convoy_realtime tests/realtime.cpp)
    target_link_libraries(convoy_realtime PRIVATE convoy)
    target_link_options(convoy_realtime PRIVATE "LINKER:--wrap=malloc,--wrap=calloc,--wrap=realloc")
    add_test(NAME realtime COMMAND convoy_realtime)
  endif()
endif()
add_executable(convoy_benchmark tests/benchmark.cpp)
target_link_libraries(convoy_benchmark PRIVATE convoy)

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS convoy EXPORT ConvoyTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} FILES_MATCHING PATTERN "*.hpp")
install(DIRECTORY vendor/dsp/include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
  FILES_MATCHING PATTERN "*.hpp")
install(FILES vendor/dsp/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/Convoy
  RENAME DSP-LICENSE)
install(EXPORT ConvoyTargets NAMESPACE convoy:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Convoy)
configure_package_config_file(cmake/ConvoyConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/ConvoyConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Convoy)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/ConvoyConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/ConvoyConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/ConvoyConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Convoy)
option(CONVOY_BUILD_EXAMPLES "Build the standalone convolution example" OFF)
if(CONVOY_BUILD_EXAMPLES)
  add_executable(convoy_example examples/convolve.cpp)
  target_link_libraries(convoy_example PRIVATE convoy::convoy)
endif()
